// Google Drive மற்றும் Mega.nz லிங்க்கை ஹேண்டில் செய்யும் முறையை மாற்றுவது window.convertToEmbedUrl = function(url) { if (!url) return ""; let trimmed = url.trim(); // 1. Google Drive Link Handle if (trimmed.includes("drive.google.com")) { let fileId = ""; let fileIdMatch = trimmed.match(/\/d\/([a-zA-Z0-9_-]+)/); if (fileIdMatch && fileIdMatch[1]) { fileId = fileIdMatch[1]; } else { let idMatch = trimmed.match(/[?&]id=([a-zA-Z0-9_-]+)/); if (idMatch && idMatch[1]) { fileId = idMatch[1]; } } if (fileId) { return `https://drive.google.com/file/d/${fileId}/preview`; } } // 2. Mega.nz Link Handle if (trimmed.includes("mega.nz")) { if (trimmed.includes("/embed/")) { return trimmed; } if (trimmed.includes("/file/")) { return trimmed.replace("/file/", "/embed/"); } if (trimmed.includes("#!")) { return trimmed.replace("#!", "embed/#!"); } return trimmed; } return trimmed; }
